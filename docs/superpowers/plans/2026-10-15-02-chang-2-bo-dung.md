```ts
import type { Canvas2D } from "@/templates/Canvas2D";

export interface DrawCall {
  op: string;
  args: unknown[];
  state: { fillStyle: unknown; font: string; globalAlpha: number; filter: string };
}

/** Context 2D giả: ghi lại mọi lệnh vẽ để test trên Node. measureText = số ký tự × cỡ chữ × 0.5. */
export class RecordingCanvas {
  readonly calls: DrawCall[] = [];
  fillStyle: unknown = "#000000";
  strokeStyle: unknown = "#000000";
  font = "10px sans-serif";
  textAlign = "start";
  textBaseline = "alphabetic";
  globalAlpha = 1;
  filter = "none";
  lineWidth = 1;
  letterSpacing = "0px";

  private record(op: string, args: unknown[]): void {
    this.calls.push({
      op,
      args,
      state: { fillStyle: this.fillStyle, font: this.font, globalAlpha: this.globalAlpha, filter: this.filter },
    });
  }

  save(): void { this.record("save", []); }
  restore(): void { this.record("restore", []); }
  fillRect(...args: unknown[]): void { this.record("fillRect", args); }
  drawImage(...args: unknown[]): void { this.record("drawImage", args); }
  fillText(...args: unknown[]): void { this.record("fillText", args); }
  beginPath(): void { this.record("beginPath", []); }
  rect(...args: unknown[]): void { this.record("rect", args); }
  roundRect(...args: unknown[]): void { this.record("roundRect", args); }
  arc(...args: unknown[]): void { this.record("arc", args); }
  fill(): void { this.record("fill", []); }
  stroke(): void { this.record("stroke", []); }
  clip(): void { this.record("clip", []); }

  measureText(text: string): { width: number } {
    const px = Number(/(\d+(?:\.\d+)?)px/.exec(this.font)?.[1] ?? 10);
    return { width: text.length * px * 0.5 };
  }

  createLinearGradient(...args: unknown[]): { stops: [number, string][]; addColorStop(o: number, c: string): void } {
    const gradient = {
      stops: [] as [number, string][],
      addColorStop(o: number, c: string) {
        gradient.stops.push([o, c]);
      },
    };
    this.record("createLinearGradient", args);
    return gradient;
  }

  asCanvas(): Canvas2D {
    return this as unknown as Canvas2D;
  }

  texts(): string[] {
    return this.calls.filter((c) => c.op === "fillText").map((c) => String(c.args[0]));
  }

  ops(name: string): DrawCall[] {
    return this.calls.filter((c) => c.op === name);
  }
}
```

```ts
import type { BrandFieldSpec, BrandKit } from "@/domain/BrandKit";
import type { SceneRole } from "@/domain/Scene";
import type { TimelineConfig } from "@/domain/Timeline";
import type { Canvas2D } from "./Canvas2D";

export interface DrawableImage {
  source: CanvasImageSource;
  width: number;
  height: number;
}

export interface FontFaceSpec {
  family: string;
  url: string;
  weight: string;
}

export interface TemplateDefaults {
  width: number;
  height: number;
  fps: number;
  timeline: TimelineConfig;
  fallbackKeyword: string;
}

export interface ProjectFrame {
  displayTitle: string;
  sourceCredit: string;
  publishDateText: string;
  brand: BrandKit;
  logo: DrawableImage | null;
}

export interface SceneFrame {
  role: SceneRole;
  text: string;
  image: DrawableImage | null;
  /** Giây tính từ đầu cảnh. */
  sceneTime: number;
  sceneDuration: number;
}

/** Một phong cách video. Hàm vẽ phải thuần theo thời gian: cùng đầu vào ra cùng hình. */
export abstract class VideoTemplate {
  abstract readonly id: string;
  abstract readonly name: string;
  abstract readonly brandFields: readonly BrandFieldSpec[];
  abstract readonly defaults: TemplateDefaults;
  abstract readonly fonts: readonly FontFaceSpec[];

  drawFrame(ctx: Canvas2D, project: ProjectFrame, scene: SceneFrame): void {
    ctx.save();
    if (scene.role === "intro") this.drawIntro(ctx, project, scene);
    else this.drawBody(ctx, project, scene);
    this.drawOverlay(ctx, project, scene);
    ctx.restore();
  }

  protected abstract drawIntro(ctx: Canvas2D, project: ProjectFrame, scene: SceneFrame): void;
  protected abstract drawBody(ctx: Canvas2D, project: ProjectFrame, scene: SceneFrame): void;
  protected drawOverlay(_ctx: Canvas2D, _project: ProjectFrame, _scene: SceneFrame): void {}
}
```

`src/templates/TemplateRegistry.ts`:
```ts
import { NotFoundError } from "@/domain/errors";
import type { VideoTemplate } from "./VideoTemplate";

export class TemplateRegistry {
  private readonly byId = new Map<string, VideoTemplate>();

  constructor(templates: readonly VideoTemplate[]) {
    for (const t of templates) {
      if (this.byId.has(t.id)) throw new Error(`Trùng phong cách ${t.id}`);
      this.byId.set(t.id, t);
    }
  }

  get(id: string): VideoTemplate {
    const t = this.byId.get(id);
    if (!t) throw new NotFoundError(`Không có phong cách ${id}`);
    return t;
  }

  list(): { id: string; name: string }[] {
    return [...this.byId.values()].map((t) => ({ id: t.id, name: t.name }));
  }
}
```

```ts
  },

  logo: { right: 96, y: 250, size: 64, gap: 14, fontPx: 46 },
  dateTag: { h: 44, padX: 16, fontPx: 26 },
  kenBurns: { from: 1, to: 1.08 },
  blurPx: 48,
} as const;

export function coverRect(srcW: number, srcH: number, box: Rect): Rect {
  const scale = Math.max(box.w / srcW, box.h / srcH);
  const w = srcW * scale;
  const h = srcH * scale;
  return { x: box.x + (box.w - w) / 2, y: box.y + (box.h - h) / 2, w, h };
}

export function kenBurnsRect(
  base: Rect,
  progress: number,
  from: number = LAYOUT.kenBurns.from,
  to: number = LAYOUT.kenBurns.to,
): Rect {
  const p = Math.min(1, Math.max(0, progress));
  const s = from + (to - from) * p;
  if (s === 1) return { ...base };
  const w = base.w * s;
  const h = base.h * s;
  return { x: base.x - (w - base.w) / 2, y: base.y - (h - base.h) / 2, w, h };
}

export function wrapText(text: string, maxWidth: number, measure: (s: string) => number, maxLines: number): string[] {
  const words = text.trim().split(/\s+/).filter(Boolean);
  if (words.length === 0) return [];
  const lines: string[] = [];
  let current = "";
  for (const word of words) {
    const candidate = current ? `${current} ${word}` : word;
    if (current && measure(candidate) > maxWidth) {
      lines.push(current);
      current = word;
    } else {
      current = candidate;
    }
  }
  if (current) lines.push(current);
  if (lines.length <= maxLines) return lines;

  const kept = lines.slice(0, maxLines - 1);
  let last = lines.slice(maxLines - 1).join(" ").split(" ");
  while (last.length > 1 && measure(`${last.join(" ")}…`) > maxWidth) last = last.slice(0, -1);
  return [...kept, `${last.join(" ")}…`];
}

export function withAlpha(hex: string, alpha: number): string {
  const n = Number.parseInt(hex.slice(1), 16);
  return `rgba(${(n >> 16) & 255},${(n >> 8) & 255},${n & 255},${alpha})`;
}
```

```ts
import type { BrandFieldSpec } from "@/domain/BrandKit";
import { DEFAULT_TIMELINE_CONFIG } from "@/domain/Timeline";
import type { Canvas2D } from "../Canvas2D";
import { VideoTemplate, type DrawableImage, type FontFaceSpec, type ProjectFrame, type SceneFrame, type TemplateDefaults } from "../VideoTemplate";
import { coverRect, kenBurnsRect, LAYOUT, withAlpha, wrapText, type Rect } from "./layout";

export const THEANH28_FONT_FAMILY = "TVS Barlow Condensed";
const DEFAULT_COLOR = "#1F9D55";
const WHITE = "#FFFFFF";

const font = (weight: number, px: number) => `${weight} ${px}px "${THEANH28_FONT_FAMILY}", "Arial Narrow", sans-serif`;
const upper = (s: string) => s.toLocaleUpperCase("vi");

export class TheAnh28Template extends VideoTemplate {
  readonly id = "theanh28";
  readonly name = "Bản tin kiểu Theanh28";

  readonly brandFields: readonly BrandFieldSpec[] = [
    { key: "channelName", label: "Tên kênh", type: "text", required: true, defaultValue: "TIN NHANH", maxLength: 24 },
    { key: "primaryColor", label: "Màu chủ đạo", type: "color", required: true, defaultValue: DEFAULT_COLOR },
    { key: "logo", label: "Logo", type: "image", required: false },
  ];

  readonly defaults: TemplateDefaults = {
    width: LAYOUT.frame.width,
    height: LAYOUT.frame.height,
    fps: 30,
    timeline: DEFAULT_TIMELINE_CONFIG,
    fallbackKeyword: "news",
  };

  readonly fonts: readonly FontFaceSpec[] = [
    { family: THEANH28_FONT_FAMILY, url: "/templates/theanh28/fonts/BarlowCondensed-SemiBold.ttf", weight: "600" },
    { family: THEANH28_FONT_FAMILY, url: "/templates/theanh28/fonts/BarlowCondensed-ExtraBold.ttf", weight: "800" },
  ];

  protected drawIntro(ctx: Canvas2D, project: ProjectFrame, scene: SceneFrame): void {
    const color = this.color(project);
    const { width, height } = LAYOUT.frame;
    const L = LAYOUT.intro;

    ctx.fillStyle = color;
    ctx.fillRect(0, 0, width, height);
    if (scene.image) this.drawClipped(ctx, scene.image, L.image, this.progress(scene));

    const gradient = ctx.createLinearGradient(0, L.gradientTop, 0, L.gradientSolid);
    gradient.addColorStop(0, withAlpha(color, 0));
    gradient.addColorStop(1, color);
    ctx.fillStyle = gradient;
    ctx.fillRect(0, L.gradientTop, width, L.gradientSolid - L.gradientTop);
    ctx.fillStyle = color;
    ctx.fillRect(0, L.gradientSolid, width, height - L.gradientSolid);

    this.drawDotPattern(ctx);
    this.drawChip(ctx, project);
    this.drawDateTag(ctx, project.publishDateText, L.dateTag.x, L.dateTag.y);

    ctx.font = font(800, L.title.fontPx);
    ctx.fillStyle = WHITE;
    ctx.textBaseline = "top";
    ctx.textAlign = "left";
    const lines = wrapText(upper(project.displayTitle), L.title.maxWidth, (s) => ctx.measureText(s).width, L.title.maxLines);
    lines.forEach((line, i) => ctx.fillText(line, L.title.x, L.title.y + i * L.title.lineHeight));
  }

  protected drawBody(ctx: Canvas2D, project: ProjectFrame, scene: SceneFrame): void {
    const color = this.color(project);
    const { width, height } = LAYOUT.frame;
    const B = LAYOUT.body;

    if (scene.image) {
      const full = coverRect(scene.image.width, scene.image.height, { x: 0, y: 0, w: width, h: height });
      const backdrop = kenBurnsRect(full, 1, 1, B.backdropScale);
      ctx.filter = `blur(${LAYOUT.blurPx}px)`;
      ctx.drawImage(scene.image.source, backdrop.x, backdrop.y, backdrop.w, backdrop.h);
      ctx.filter = "none";
      ctx.fillStyle = `rgba(0,0,0,${B.backdropShade})`;
      ctx.fillRect(0, 0, width, height);
      this.drawClipped(ctx, scene.image, B.band, this.progress(scene));
    } else {
      const gradient = ctx.createLinearGradient(0, 0, 0, height);
      gradient.addColorStop(0, color);
      gradient.addColorStop(1, "#0B0F0D");
      ctx.fillStyle = gradient;
      ctx.fillRect(0, 0, width, height);
    }
    this.drawDateTag(ctx, project.publishDateText, B.dateTag.x, B.dateTag.y);
  }

  protected drawOverlay(ctx: Canvas2D, project: ProjectFrame, scene: SceneFrame): void {
    const C = LAYOUT.credit;
    ctx.font = font(600, C.fontPx);
    ctx.fillStyle = WHITE;
    ctx.textBaseline = "top";
    ctx.textAlign = "left";
    const lines = project.sourceCredit ? [`THEO: ${upper(project.sourceCredit)}`, "ẢNH: TỔNG HỢP"] : ["ẢNH: TỔNG HỢP"];
    lines.forEach((line, i) => ctx.fillText(line, C.x, C.y + i * C.lineHeight));
    if (scene.role === "body") this.drawCornerLogo(ctx, project);
  }

  private color(project: ProjectFrame): string {
    return project.brand.get("primaryColor") ?? DEFAULT_COLOR;
  }

  private channelName(project: ProjectFrame): string {
    return upper(project.brand.get("channelName") ?? "TIN NHANH");
  }

  private progress(scene: SceneFrame): number {
    return scene.sceneDuration > 0 ? scene.sceneTime / scene.sceneDuration : 0;
  }

  private drawClipped(ctx: Canvas2D, image: DrawableImage, box: Rect, progress: number): void {
    const target = kenBurnsRect(coverRect(image.width, image.height, box), progress);
    ctx.save();
    ctx.beginPath();
    ctx.rect(box.x, box.y, box.w, box.h);
    ctx.clip();
    ctx.drawImage(image.source, target.x, target.y, target.w, target.h);
    ctx.restore();
  }

  private drawDotPattern(ctx: Canvas2D): void {
    const P = LAYOUT.intro.pattern;
    ctx.save();
    ctx.globalAlpha = P.alpha;
    ctx.fillStyle = WHITE;
    ctx.beginPath();
    for (let y = P.y + P.spacing / 2; y < P.y + P.h; y += P.spacing) {
      for (let x = P.x + P.spacing / 2; x < P.x + P.w; x += P.spacing) {
        ctx.rect(x - P.radius, y - P.radius, P.radius * 2, P.radius * 2);
      }
    }
    ctx.fill();
    ctx.restore();
  }

  private drawLogoBadge(ctx: Canvas2D, project: ProjectFrame, x: number, y: number, size: number): void {
    const color = this.color(project);
    ctx.save();
    ctx.beginPath();
    ctx.arc(x + size / 2, y + size / 2, size / 2, 0, Math.PI * 2);
    ctx.clip();
    if (project.logo) {
      const r = coverRect(project.logo.width, project.logo.height, { x, y, w: size, h: size });
      ctx.drawImage(project.logo.source, r.x, r.y, r.w, r.h);
    } else {
      ctx.fillStyle = WHITE;
      ctx.fillRect(x, y, size, size);
      ctx.fillStyle = color;
      ctx.font = font(800, Math.round(size * 0.55));
      ctx.textAlign = "center";
      ctx.textBaseline = "middle";
      ctx.fillText(this.channelName(project).charAt(0), x + size / 2, y + size / 2);
    }
    ctx.restore();
  }

  private drawChip(ctx: Canvas2D, project: ProjectFrame): void {
    const K = LAYOUT.intro.chip;
    const name = this.channelName(project);
    ctx.font = font(800, K.fontPx);
    const textW = ctx.measureText(name).width;
    const w = K.padX + K.logoSize + K.gap + textW + K.padX;
    ctx.fillStyle = this.color(project);
    ctx.strokeStyle = WHITE;
    ctx.lineWidth = 3;
    ctx.beginPath();
    ctx.roundRect(K.x, K.y, w, K.h, K.h / 2);
    ctx.fill();
    ctx.stroke();
    this.drawLogoBadge(ctx, project, K.x + K.padX, K.y + (K.h - K.logoSize) / 2, K.logoSize);
    ctx.font = font(800, K.fontPx);
    ctx.fillStyle = WHITE;
    ctx.textAlign = "left";
    ctx.textBaseline = "middle";
    ctx.fillText(name, K.x + K.padX + K.logoSize + K.gap, K.y + K.h / 2);
  }

  private drawDateTag(ctx: Canvas2D, dateText: string, x: number, y: number): void {
    const D = LAYOUT.dateTag;
    const text = `NGÀY ĐĂNG: ${dateText}`;
    ctx.font = font(600, D.fontPx);
    const w = ctx.measureText(text).width + D.padX * 2;
    ctx.fillStyle = "rgba(0,0,0,0.45)";
    ctx.fillRect(x, y, w, D.h);
    ctx.fillStyle = WHITE;
    ctx.textAlign = "left";
    ctx.textBaseline = "middle";
    ctx.fillText(text, x + D.padX, y + D.h / 2);
  }

  private drawCornerLogo(ctx: Canvas2D, project: ProjectFrame): void {
    const G = LAYOUT.logo;
    const name = this.channelName(project);
    ctx.font = font(800, G.fontPx);
    const textW = ctx.measureText(name).width;
    const textX = LAYOUT.frame.width - G.right - textW;
    this.drawLogoBadge(ctx, project, textX - G.gap - G.size, G.y, G.size);
    ctx.font = font(800, G.fontPx);
    ctx.fillStyle = WHITE;
    ctx.textAlign = "left";
    ctx.textBaseline = "middle";
    ctx.fillText(name, textX, G.y + G.size / 2);
  }
}
```

```ts
export const MIX_DEFAULTS = { voiceTargetDb: -16.5, gateDb: -45, fadeInSec: 1.5, fadeOutSec: 2, ceiling: 0.89 } as const;

export interface MonoClip {
  startSec: number;
  samples: Float32Array;
}

export interface MusicBed {
  /** 1 kênh (mono) hoặc 2 kênh (trái, phải), cùng sampleRate với bản trộn. */
  channels: Float32Array[];
  levelBelowVoiceDb: number;
}

export interface MixOptions {
  sampleRate: number;
  durationSec: number;
  voiceTargetDb?: number;
  fadeInSec?: number;
  fadeOutSec?: number;
  ceiling?: number;
}

export interface StereoMix {
  left: Float32Array;
  right: Float32Array;
  sampleRate: number;
}

export function dbToGain(db: number): number {
  return 10 ** (db / 20);
}

/** RMS (dB) của các mẫu có biên độ vượt cổng; -Infinity nếu không có mẫu nào. */
export function rmsDb(samples: Float32Array, gateDb: number = MIX_DEFAULTS.gateDb): number {
  const gate = dbToGain(gateDb);
  let sum = 0;
  let count = 0;
  for (const s of samples) {
    if (Math.abs(s) >= gate) {
      sum += s * s;
      count++;
    }
  }
  return count === 0 ? -Infinity : 10 * Math.log10(sum / count);
}

function fadeGain(i: number, n: number, sampleRate: number, fadeInSec: number, fadeOutSec: number): number {
  const fin = Math.round(fadeInSec * sampleRate);
  const fout = Math.round(fadeOutSec * sampleRate);
  let g = 1;
  if (fin > 0 && i < fin) g = Math.min(g, i / fin);
  if (fout > 0 && i >= n - fout) g = Math.min(g, (n - 1 - i) / fout);
  return Math.max(0, g);
}

export function mixProgram(voices: readonly MonoClip[], music: MusicBed | null, options: MixOptions): StereoMix {
  const sr = options.sampleRate;
  const n = Math.ceil(options.durationSec * sr);
  const target = options.voiceTargetDb ?? MIX_DEFAULTS.voiceTargetDb;
  const fadeIn = options.fadeInSec ?? MIX_DEFAULTS.fadeInSec;
  const fadeOut = options.fadeOutSec ?? MIX_DEFAULTS.fadeOutSec;
  const ceiling = options.ceiling ?? MIX_DEFAULTS.ceiling;

  const voice = new Float32Array(n);
  for (const clip of voices) {
    const start = Math.round(clip.startSec * sr);
    for (let j = 0; j < clip.samples.length && start + j < n; j++) {
      voice[start + j] = (voice[start + j] ?? 0) + (clip.samples[j] ?? 0);
    }
  }
  const voiceLevel = rmsDb(voice);
  const voiceGain = Number.isFinite(voiceLevel) ? dbToGain(target - voiceLevel) : 1;

  const left = new Float32Array(n);
  const right = new Float32Array(n);
  for (let i = 0; i < n; i++) {
    const v = (voice[i] ?? 0) * voiceGain;
    left[i] = v;
    right[i] = v;
  }

  if (music && music.channels.length > 0) {
    const srcL = music.channels[0] as Float32Array;
    const srcR = music.channels[1] ?? srcL;
    const musicLevel = rmsDb(srcL, -60);
    if (Number.isFinite(musicLevel) && srcL.length > 0) {
      const gain = dbToGain(target - music.levelBelowVoiceDb - musicLevel);
      for (let i = 0; i < n; i++) {
        const f = gain * fadeGain(i, n, sr, fadeIn, fadeOut);
        left[i] = (left[i] ?? 0) + (srcL[i % srcL.length] ?? 0) * f;
        right[i] = (right[i] ?? 0) + (srcR[i % srcR.length] ?? 0) * f;
      }
    }
  }

  let peak = 0;
  for (let i = 0; i < n; i++) peak = Math.max(peak, Math.abs(left[i] ?? 0), Math.abs(right[i] ?? 0));
  if (peak > ceiling) {
    const k = ceiling / peak;
    for (let i = 0; i < n; i++) {
      left[i] = (left[i] ?? 0) * k;
      right[i] = (right[i] ?? 0) * k;
    }
  }
  return { left, right, sampleRate: sr };
}
```

```ts
/** Thời điểm (giây) cần lấy khung từ cảnh trám; null khi cảnh không có hình. */
export function sourceTimeFor(playback: FootagePlayback, sceneTime: number): number | null {
  const src = playback.sourceDurationSec ?? 0;
  switch (playback.mode) {
    case "none":
      return null;
    case "still":
      return 0;
    case "clip":
      return Math.min(sceneTime, Math.max(0, src - 0.001));
    case "loop":
      return src > 0 ? sceneTime % src : 0;
  }
}
```

```ts
import type { FootagePlayback } from "@/domain/Footage";
import type { SceneRole } from "@/domain/Scene";
import type { Timeline } from "@/domain/Timeline";
import { sourceTimeFor } from "@/client/media/sourceTime";
import type { Canvas2D } from "@/templates/Canvas2D";
import type { DrawableImage, ProjectFrame, VideoTemplate } from "@/templates/VideoTemplate";
import { RenderCancelledError } from "./errors";

export interface FrameProvider {
  frameAt(sourceTime: number): Promise<DrawableImage | null>;
  dispose(): void;
}

export interface RenderScene {
  sceneId: string;
  role: SceneRole;
  text: string;
  playback: FootagePlayback;
  frames: FrameProvider | null;
}

export interface VideoFrameSink {
  /** Mã hoá nội dung canvas hiện tại thành một khung. */
  add(timestampSec: number, durationSec: number): Promise<void>;
}

export interface RenderProgress {
  frame: number;
  totalFrames: number;
}

export interface VideoRenderInput {
  template: VideoTemplate;
  ctx: Canvas2D;
  project: ProjectFrame;
  timeline: Timeline;
  scenes: ReadonlyMap<string, RenderScene>;
  fps: number;
  sink: VideoFrameSink;
  onProgress?: (progress: RenderProgress) => void;
  signal?: AbortSignal;
}

const PROGRESS_EVERY = 15;

export class VideoRenderer {
  constructor(private readonly input: VideoRenderInput) {}

  async render(): Promise<number> {
    const { template, ctx, project, timeline, scenes, fps, sink, onProgress, signal } = this.input;
    const total = timeline.frameCount(fps);
    for (let i = 0; i < total; i++) {
      if (signal?.aborted) throw new RenderCancelledError();
      const t = i / fps;
      const slot = timeline.slotAt(t);
      const scene = scenes.get(slot.sceneId);
      if (!scene) throw new Error(`Thiếu dữ liệu cảnh ${slot.sceneId}`);
      const sceneTime = t - slot.start;
      const sourceTime = sourceTimeFor(scene.playback, sceneTime);
      const image = sourceTime === null || !scene.frames ? null : await scene.frames.frameAt(sourceTime);
      template.drawFrame(ctx, project, {
        role: scene.role,
        text: scene.text,
        image,
        sceneTime,
        sceneDuration: slot.end - slot.start,
      });
      await sink.add(t, 1 / fps);
      const frame = i + 1;
      if (frame % PROGRESS_EVERY === 0 || frame === total) onProgress?.({ frame, totalFrames: total });
    }
    return total;
  }
}
```

```ts
  async decode(bytes: ArrayBuffer): Promise<DecodedAudio> {

    const buffer = await ctx.decodeAudioData(bytes.slice(0));
    const channels = Array.from({ length: buffer.numberOfChannels }, (_, i) => buffer.getChannelData(i).slice());

  static toMono(channels: Float32Array[]): Float32Array {
    const first = channels[0];
    if (!first) return new Float32Array();
    if (channels.length === 1) return first;
    const out = new Float32Array(first.length);
    for (const ch of channels) for (let i = 0; i < out.length; i++) out[i] = (out[i] ?? 0) + (ch[i] ?? 0) / channels.length;
```

```ts
    readonly width: number,
    readonly height: number,

  static async open(blob: Blob, maxSide = 1080): Promise<VideoFrameProvider> {
    const input = new Input({ formats: ALL_FORMATS, source: new BlobSource(blob) });
    const track = await input.getPrimaryVideoTrack();
    if (!track) {
      input.dispose();
      throw new Error("File video không có hình");
    }
    const w = await track.getCodedWidth();
    const h = await track.getCodedHeight();
    const scale = Math.min(1, maxSide / Math.max(w, h));
    const width = Math.round(w * scale);
    const height = Math.round(h * scale);
    const sink = new CanvasSink(track, { width, height, fit: "fill" });
    const durationSec = await input.computeDuration();
    return new VideoFrameProvider(input, sink, durationSec, width, height);
  }

  async frameAt(sourceTime: number): Promise<DrawableImage | null> {
    if (this.iterator === null || sourceTime < this.lastTime) {
      await this.iterator?.return();
      this.iterator = this.sink.canvases(sourceTime);
      this.current = null;
      const first = await this.iterator.next();
      this.upcoming = first.done ? null : first.value;
    }
    this.lastTime = sourceTime;
    while (this.upcoming && this.upcoming.timestamp <= sourceTime + 1e-6) {
      this.current = this.upcoming;
      const next = await this.iterator.next();
      this.upcoming = next.done ? null : next.value;
    }
    const frame = this.current ?? this.upcoming;
    return frame ? { source: frame.canvas, width: frame.canvas.width, height: frame.canvas.height } : null;
  }

  dispose(): void {
    void this.iterator?.return();
    this.input.dispose();
  }
```

```ts
import type { StereoMix } from "@/client/media/AudioMix";
import type { VideoFrameSink } from "./VideoRenderer";

export interface Mp4EncoderOptions {
  canvas: OffscreenCanvas;
  fps: number;
  videoBitrate: number;
  audioBitrate: number;
}

const AUDIO_CHUNK_SEC = 1;

export class Mp4Encoder implements VideoFrameSink {
  private readonly target = new BufferTarget();
  private readonly output: Output;
  private readonly video: CanvasSource;
  private readonly audio: AudioSampleSource;

  constructor(options: Mp4EncoderOptions) {
    this.output = new Output({ format: new Mp4OutputFormat({ fastStart: "in-memory" }), target: this.target });
    this.video = new CanvasSource(options.canvas, { codec: "avc", bitrate: options.videoBitrate, keyFrameInterval: 2 });
    this.audio = new AudioSampleSource({ codec: "aac", bitrate: options.audioBitrate });
    this.output.addVideoTrack(this.video, { frameRate: options.fps });
    this.output.addAudioTrack(this.audio);
  }

  start(): Promise<void> {
    return this.output.start();
  }

  add(timestampSec: number, durationSec: number): Promise<void> {
    return this.video.add(timestampSec, durationSec);
  }

  async addAudio(mix: StereoMix): Promise<void> {
    const chunk = Math.round(AUDIO_CHUNK_SEC * mix.sampleRate);
    for (let start = 0; start < mix.left.length; start += chunk) {
      const frames = Math.min(chunk, mix.left.length - start);
      const data = new Float32Array(frames * 2);
      data.set(mix.left.subarray(start, start + frames), 0);
      data.set(mix.right.subarray(start, start + frames), frames);
      const sample = new AudioSample({
        data,
        format: "f32-planar",
        numberOfChannels: 2,
        sampleRate: mix.sampleRate,
        timestamp: start / mix.sampleRate,
      });
      await this.audio.add(sample);
      sample.close();
    }
  }

  async finish(): Promise<Blob> {
    this.video.close();
    this.audio.close();
    await this.output.finalize();
    if (!this.target.buffer) throw new Error("Không tạo được file MP4");
    return new Blob([this.target.buffer], { type: "video/mp4" });
  }

  cancel(): Promise<void> {
    return this.output.cancel();
  }
}
```

```ts
import { RenderCancelledError } from "./errors";
import type { RenderJob, WorkerMessage, WorkerRequest } from "./RenderJob";

export interface WorkerLike {
  onmessage: ((e: { data: WorkerMessage }) => void) | null;
  onerror: ((e: { message?: string }) => void) | null;
  postMessage(message: WorkerRequest): void;
  terminate(): void;
}

type Progress = Extract<WorkerMessage, { type: "progress" }>;

const defaultWorker = (): WorkerLike =>
  new Worker(new URL("./render.worker.ts", import.meta.url), { type: "module" }) as unknown as WorkerLike;

export class RenderClient {
  private active: { worker: WorkerLike; reject: (e: Error) => void } | null = null;

  constructor(private readonly createWorker: () => WorkerLike = defaultWorker) {}

  render(job: RenderJob, onProgress?: (p: Progress) => void): Promise<{ blob: Blob; durationSec: number }> {
    return new Promise((resolve, reject) => {
      const worker = this.createWorker();
      const finish = () => {
        worker.terminate();
        this.active = null;
      };
      this.active = { worker, reject };
      worker.onmessage = (e) => {
        const m = e.data;
        if (m.type === "progress") onProgress?.(m);
        else if (m.type === "done") {
          finish();
          resolve({ blob: m.blob, durationSec: m.durationSec });
        } else {
          finish();
          reject(new Error(m.message));
        }
      };
      worker.onerror = (e) => {
        finish();
        reject(new Error(`Lỗi bộ dựng: ${e.message ?? "không rõ"}`));
      };
      worker.postMessage({ type: "start", job });
    });
  }

  cancel(): void {
    if (!this.active) return;
    const { worker, reject } = this.active;
    worker.terminate();
    this.active = null;
    reject(new RenderCancelledError());
  }
}
```

```ts
import { Mp4Encoder } from "./Mp4Encoder";
import { planRender } from "./RenderPlan";
import type { FootageInput, RenderJob, WorkerMessage, WorkerRequest } from "./RenderJob";
import { VideoRenderer, type FrameProvider, type RenderScene } from "./VideoRenderer";

interface WorkerScope {
  onmessage: ((e: MessageEvent<WorkerRequest>) => void) | null;
  postMessage(message: WorkerMessage): void;
  fonts: FontSink;
}

const scope = globalThis as unknown as WorkerScope;
const VIDEO_BITRATE = 8_000_000;
const AUDIO_BITRATE = 192_000;

async function blobOf(footage: FootageInput, origin: string): Promise<Blob> {
  if (footage.blob) return footage.blob;
  if (!footage.url) throw new Error("Cảnh trám thiếu nguồn");
  const response = await fetch(new URL(footage.url, origin), { mode: "cors" });
  if (!response.ok) throw new Error(`Không tải được cảnh trám (${response.status})`);
  return response.blob();
}

async function render(job: RenderJob): Promise<void> {
  scope.postMessage({ type: "progress", stage: "loading" });
  const template = defaultTemplateRegistry().get(job.templateId);
  await loadFonts(template.fonts, scope.fonts, job.origin);
  const brand = BrandKit.create(template.id, template.brandFields, job.brandValues);
  const logoBitmap = job.logo ? await createImageBitmap(job.logo) : null;
  const { timeline, voices } = planRender(job, template.defaults.timeline);

  const providers: FrameProvider[] = [];
  const scenes = new Map<string, RenderScene>();
  try {
    for (const [index, s] of job.scenes.entries()) {
      const slot = timeline.slots[index];
      if (!slot) throw new Error(`Thiếu mốc thời gian cho cảnh ${s.sceneId}`);
      const sceneDuration = slot.end - slot.start;
      if (s.footage.kind === "fallback") {
        scenes.set(s.sceneId, { sceneId: s.sceneId, role: s.role, text: s.text, playback: { mode: "none" }, frames: null });
        continue;
      }
      const blob = await blobOf(s.footage, job.origin);
      if (s.footage.kind === "image") {
        const frames = await ImageFrameProvider.fromBlob(blob);
        providers.push(frames);
        scenes.set(s.sceneId, { sceneId: s.sceneId, role: s.role, text: s.text, playback: { mode: "still" }, frames });
      } else {
        const frames = await VideoFrameProvider.open(blob);
        providers.push(frames);
        const footage = new VideoFootage(s.sceneId, { origin: "upload", uri: "blob" }, frames.width, frames.height, frames.durationSec);
        scenes.set(s.sceneId, {
          sceneId: s.sceneId,
          role: s.role,
          text: s.text,
          playback: footage.playbackFor(sceneDuration),
          frames,
        });
      }
    }

    scope.postMessage({ type: "progress", stage: "audio" });
    const mix = mixProgram(voices, job.music, { sampleRate: job.sampleRate, durationSec: timeline.totalDuration });

    const { width, height, fps } = template.defaults;
    const canvas = new OffscreenCanvas(width, height);
    const ctx = canvas.getContext("2d");
    if (!ctx) throw new Error("Trình duyệt không cấp được canvas 2D");
    const encoder = new Mp4Encoder({ canvas, fps, videoBitrate: VIDEO_BITRATE, audioBitrate: AUDIO_BITRATE });
    await encoder.start();
    await encoder.addAudio(mix);

    await new VideoRenderer({
      template,

      project: {
        ...job.project,
        brand,
        logo: logoBitmap ? { source: logoBitmap, width: logoBitmap.width, height: logoBitmap.height } : null,
      },
      timeline,
      scenes,
      fps,
      sink: encoder,
      onProgress: (p) => scope.postMessage({ type: "progress", stage: "video", frame: p.frame, totalFrames: p.totalFrames }),
    }).render();

    scope.postMessage({ type: "progress", stage: "finalizing" });
    const blob = await encoder.finish();
    scope.postMessage({ type: "done", blob, durationSec: timeline.totalDuration });
  } finally {
    for (const p of providers) p.dispose();
    logoBitmap?.close();
  }
}

scope.onmessage = (event) => {
  if (event.data.type !== "start") return;
  render(event.data.job).catch((error: unknown) => {
    scope.postMessage({ type: "error", message: error instanceof Error ? error.message : "Dựng video thất bại" });
  });
};
```

```ts
import { ApiError } from "./AuthClient";

export interface VoiceInfo {
  name: string;
  gender: string;
  tier: string;
}

export interface TtsApi {
  voices(): Promise<VoiceInfo[]>;
  synthesize(text: string, voiceName: string, speakingRate: number): Promise<ArrayBuffer>;
}

async function toApiError(response: Response): Promise<ApiError> {
  const body = (await response.json().catch(() => null)) as { error?: { code?: string; message?: string } } | null;
  return new ApiError(response.status, body?.error?.code ?? "INTERNAL", body?.error?.message ?? "Lỗi máy chủ");
}

export class TtsClient implements TtsApi {
  constructor(private readonly fetchImpl: typeof fetch = (...args) => fetch(...args)) {}

  async voices(): Promise<VoiceInfo[]> {
    const response = await this.fetchImpl("/api/tts/voices");
    if (!response.ok) throw await toApiError(response);
    return ((await response.json()) as { voices: VoiceInfo[] }).voices;
  }

  async synthesize(text: string, voiceName: string, speakingRate: number): Promise<ArrayBuffer> {
    const response = await this.fetchImpl("/api/tts", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ text, voiceName, speakingRate }),
    });
    if (!response.ok) throw await toApiError(response);
    return response.arrayBuffer();
  }
}
```

```ts
  constructor(private readonly deps: DevRenderDeps) {}

  get state(): DevRenderState {
    return this.current;

    this.listeners.add(listener);
    return () => this.listeners.delete(listener);
```

- Tốc độ dựng phụ thuộc máy; video 1 phút trên máy yếu có thể quá 2 phút. Nếu quá chậm, giảm cỡ giải mã video cảnh trám (`maxSide`) hoặc bỏ nền mờ cho cảnh video.
- `ctx.filter = "blur()"` trên OffscreenCanvas được Chrome/Edge hỗ trợ; Safari có thể bỏ qua (nền không mờ, không lỗi).
- Bộ giải mã AAC/H.264 phần cứng có thể từ chối 1080×1920 trên máy rất cũ; `BrowserSupport` báo trước khi dựng.
- `FakeStockProvider` vẫn trả SVG data URI (createImageBitmap không đọc được SVG blob trong Chrome); chặng 3 phải đổi sang ảnh PNG/JPEG hoặc dùng URL CDN thật khi chạy giả lập.
